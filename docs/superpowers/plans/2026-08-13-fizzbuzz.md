```python
import io
import unittest
from contextlib import redirect_stdout

from fizzbuzz import fizzbuzz


class FizzBuzzTests(unittest.TestCase):
    def test_returns_and_prints_standard_sequence(self):
        output = io.StringIO()
        with redirect_stdout(output):
            result = fizzbuzz(15)

        expected = [
            1, 2, "Fizz", 4, "Buzz", "Fizz", 7, 8, "Fizz", "Buzz",
            11, "Fizz", 13, 14, "FizzBuzz",
        ]
        self.assertEqual(result, expected)
        self.assertEqual(output.getvalue(), "\n".join(map(str, expected)) + "\n")

    def test_rejects_non_positive_input(self):
        with self.assertRaises(ValueError):
            fizzbuzz(0)


if __name__ == "__main__":
    unittest.main()
```

### Task 2: Implement and document FizzBuzz

```python
def fizzbuzz(n):
    """Print and return the FizzBuzz sequence from 1 through n.

    Args:
        n: Positive integer marking the end of the sequence.

    Returns:
        A list containing integers and the strings "Fizz", "Buzz", or
        "FizzBuzz".

    Raises:
        ValueError: If n is not a positive integer.
    """
    if not isinstance(n, int) or isinstance(n, bool) or n <= 0:
        raise ValueError("n must be a positive integer")

    results = []
    for number in range(1, n + 1):
        if number % 15 == 0:
            result = "FizzBuzz"
        elif number % 3 == 0:
            result = "Fizz"
        elif number % 5 == 0:
            result = "Buzz"
        else:
            result = number
        results.append(result)
        print(result)
    return results


if __name__ == "__main__":
    fizzbuzz(15)
```
